  % surface: z = x^2 + y^2 paraboloid sketch
  \draw[thick] (0,2.6) parabola bend (0,0.2) (3.0,2.6);
  \draw[thick] (0.5,3.1) parabola bend (0.5,0.7) (3.5,3.1);
  % slice plane at y = const
  \draw[thick,dashed,accent] (0.9,0.2) -- (0.9,2.9);
  \draw[thick,accent] (0.9,0.55) parabola bend (2.0,0.55) (3.1,2.75);
  % tangent slope on slice
  \draw[->,very thick] (2.0,0.55) -- ++(0.9,0.95) node[right,font=\footnotesize] {slope $=\partial f/\partial x$};
  \draw[dotted,gray] (2.0,0.55) -- ++(0.9,0);
  \node[font=\footnotesize,align=center] at (1.6,3.4) {fix $y$: the surface becomes a curve,\\and the partial is its ordinary slope};
\end{tikzpicture}
