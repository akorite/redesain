% sample "good" proof for the linter demo.
\begin{theorem}
For every integer n, n^2 + n is even.
\end{theorem}
\begin{lemma}[Parity split]
Every integer is even or odd.
\end{lemma}
\begin{proof}
If n is even, n^2+n is a sum of evens; if n is odd, n^2 is odd and
n is odd, so their sum is even. \qed
\end{proof}
