% fig03-growth.tex -- Figure 1.3: operation-count growth vs the 1 s budget.
% Log-log axes; the 1e8-op line marks one second of simple work.
\begin{tikzpicture}
\begin{loglogaxis}[
    width=0.70\linewidth, height=6.6cm,
    xlabel={input size $n$}, ylabel={operations},
    xmin=1, xmax=1e8, ymin=1, ymax=1e17,
    xtick={1,1e2,1e4,1e6,1e8}, ytick={1,1e4,1e8,1e12,1e16},
    legend pos=outer north east, legend cell align=left, legend style={font=\footnotesize, draw=rulegray, fill=white},
    grid=major, grid style={gray!22}, axis line style={slate},
    tick label style={font=\footnotesize},
    label style={font=\small}]
\addplot[domain=1:1e8, samples=60, thick, good] {ln(x)/ln(2)};
\addplot[domain=1:1e8, samples=60, thick, ink] {x};
\addplot[domain=1:1e8, samples=80, thick, accent, dashed] {x*ln(x)/ln(2)};
\addplot[domain=1:1e8, samples=60, thick, amber, dotted] {x^2};
\addplot[domain=1:56,   samples=57, thick, ink, dashdotted] {2^x};
\addplot[domain=1:1e8, samples=2, very thick, amber!70] {1e8};
\legend{$\log_2 n$, $n$, $n\log_2 n$, $n^2$, $2^n$, 1-second budget ($10^8$)}
\end{loglogaxis}
\end{tikzpicture}
