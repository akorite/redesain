% fig74-rhash.tex -- Figure 14.x: rolling hash on "banana": slide the window from "ana" to "nan".
\input{figures/fstyle}
\begin{tikzpicture}[font=\footnotesize]
\foreach \c [count=\i from 0] in {b,a,n,a,n,a}
  \node[\ifnum\i<1 cell\else\ifnum\i>3 cell\else cellh\fi\fi, minimum width=7mm] (s\i) at (\i*0.78,0) {\texttt{\c}};
\draw[decorate,decoration={brace,raise=3pt},accent,thick] (s1.north west) -- (s3.north east) node[midway,above=5pt,labA]{window 1: \texttt{ana}};
\draw[decorate,decoration={brace,mirror,raise=3pt},amber,thick] (s2.south west) -- (s4.south east) node[midway,below=5pt,labM]{window 2: \texttt{nan}};
\node[lab, anchor=west, align=left] at (5.2,0.9) {$h(\texttt{ana})=\texttt{a}B^2+\texttt{n}B+\texttt{a}$};
\node[lab, anchor=west, align=left] at (5.2,0.0) {slide: drop the leading \texttt{a}, shift by $B$, append \texttt{n}};
\node[labM, anchor=west, align=left] at (5.2,-0.9) {$h(\texttt{nan})=\big(h(\texttt{ana})-\texttt{a}B^2\big)B+\texttt{n}\pmod p$};
\node[lab, anchor=west] at (0,-1.9) {$O(1)$ per slide, so all $n-k+1$ window hashes cost $O(n)$};
\end{tikzpicture}
