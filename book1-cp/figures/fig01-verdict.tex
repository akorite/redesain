% fig01-verdict.tex -- Figure 1.1: what the judge does with a submission.
\input{figures/fstyle}
\fitfig{\begin{tikzpicture}[
  font=\small,
  box/.style={draw=ink, rounded corners=1.5pt, minimum height=8mm,
              inner xsep=7pt, fill=graybg},
  good/.style={draw=good, rounded corners=1.5pt, minimum height=8mm,
              minimum width=2.2cm, fill=goodbg, text=good},
  bad/.style={draw=amber, rounded corners=1.5pt, minimum height=8mm,
              minimum width=2.2cm, fill=amberbg, text=amber},
  lab/.style={font=\footnotesize, text=slate, fill=white, inner sep=1.5pt},
  arr/.style={-{stealth}, thick}
]
\node[box] (sub) at (0,0) {source code};
\node[box, right=9mm of sub] (comp) {compile};
\node[box, right=9mm of comp] (run) {run on hidden tests};
\node[good] (ac) at ($(run.east)+(3.6,0.9)$) {Accepted};
\node[bad]  (rej) at ($(run.east)+(3.6,-0.9)$) {Rejected};
\draw[arr] (sub) -- (comp);
\draw[arr] (comp) -- (run);
\draw[arr] (run.east) -- (ac.west) node[lab, pos=0.45, above=1pt] {every test passed};
\draw[arr] (run.east) -- (rej.west) node[lab, pos=0.45, below=1pt] {any failure};
\draw[arr, amber] (comp.south) -- ++(0,-1.5) -| (rej.south)
  node[lab, pos=0.25, below=1pt, text=amber] {does not compile};
\node[font=\footnotesize, text width=0.92\linewidth, align=center, text=slate]
  at ($(run.south)+(1.2,-2.5)$)
  {Rejected comes in five flavours: \textbf{WA} wrong answer,
   \textbf{TLE} too slow, \textbf{RE} crash, \textbf{MLE} too much memory,
   \textbf{CE} never built};
\end{tikzpicture}}
