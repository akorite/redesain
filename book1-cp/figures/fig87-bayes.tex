% fig87-bayes.tex -- Figure 16.x: Bayes by a probability tree; 1% prevalence, 99% sensitivity, 5% false positives -> P(sick|+) = 16.7%.
\input{figures/fstyle}
\begin{tikzpicture}[font=\footnotesize]
\tikzset{tb/.style={draw=slate, rounded corners=2pt, inner sep=3pt, fill=white}}
\node[tb] (r) at (0,0) {population};
\node[tb, draw=amber] (s) at (3.4,1.6) {sick $1\%$};
\node[tb, draw=accent] (h) at (3.4,-1.6) {healthy $99\%$};
\node[tb] (pp) at (7.2,2.3) {test$+$ $99\%$\ $\to$\ $0.0099$};
\node[tb] (pn) at (7.2,1.0) {test$-$ $1\%$\ $\to$\ $0.0001$};
\node[tb] (hp) at (7.2,-1.0) {test$+$ $5\%$\ $\to$\ $0.0495$};
\node[tb] (hn) at (7.2,-2.3) {test$-$ $95\%$\ $\to$\ $0.9405$};
\draw[arr] (r)--(s); \draw[arr] (r)--(h); \draw[arr] (s)--(pp); \draw[arr] (s)--(pn); \draw[arr] (h)--(hp); \draw[arr] (h)--(hn);
\draw[amber, very thick] (pp.south west) rectangle (pp.north east); \draw[accent, very thick] (hp.south west) rectangle (hp.north east);
\node[labM, anchor=west, align=left] at (0,-3.6) {$\Pr[\text{sick}\mid +]=\dfrac{0.0099}{0.0099+0.0495}=\dfrac{0.0099}{0.0594}\approx16.7\%$};
\end{tikzpicture}
